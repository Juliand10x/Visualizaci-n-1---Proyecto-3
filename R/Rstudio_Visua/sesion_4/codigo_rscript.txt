# 1. Definir la ruta directa al Shapefile de Localidades
ruta_shp <- "C:/Users/santi/OneDrive - Universidad Externado de Colombia/Ciencia_Datos_UEC/VisualizaciónI-CD/Sesion_04_PowerBI_RStudio/Localidades/LocalidadesBogota.shp"

# 2. Cargar la capa cartográfica sf
localidades_sf <- st_read(ruta_shp, quiet = TRUE)

# 3. Estandarizar nombres en el Shapefile y en el dataset autogenerado por Power BI
localidades_sf <- localidades_sf %>%
  rename(localidad = NOMBRE_COM) %>%
  mutate(localidad = trimws(toupper(as.character(localidad))))

dataset_limpio <- dataset %>%
  mutate(localidad = trimws(toupper(as.character(localidad))))

# 4. Realizar la unión relacional espacial
mapa_final <- localidades_sf %>%
  left_join(dataset_limpio, by = "localidad")

# 5. Configurar tmap en modo de gráfico estático para Power BI
tmap_mode("plot")

# 6. Construir el mapa choropleth con clasificación K-Means
mapa_kmeans_ingresos <- tm_shape(mapa_final) +
  tm_polygons(
    col = "N_ingpcugaj13",
    title = "Ingreso Promedio",
    palette = "Purples",
    style = "kmeans",            # Clasificación K-Means
    n = 5,                       # 5 intervalos
    border.col = "white",
    lwd = 0.5
  ) +
  tm_layout(
    main.title = "Distribución del Ingreso por Localidades (K-Means)",
    main.title.size = 1.0,
    legend.outside = TRUE,
    legend.outside.position = "right",
    frame = FALSE,
    outer.margins = c(0, 0, 0, 0)
  )

# 7. Renderizar el objeto tmap
print(mapa_kmeans_ingresos)
